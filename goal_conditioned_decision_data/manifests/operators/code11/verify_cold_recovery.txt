def require(condition, message):
    if not condition:
        raise RuntimeError(message)


def check_time():
    if time.monotonic() >= DEADLINE:
        raise TimeoutError('inclusive cold verification deadline reached')


def guarded_local(root, relative, *args, **kwargs):
    if str(relative).startswith(BULK):
        attempts['payload_path'] += 1
        raise RuntimeError('bulk local payload read refused')
    check_time()
    return raw_local(root, relative, *args, **kwargs)


def guarded_read(root, relative, *args, **kwargs):
    if str(relative).startswith(BULK):
        attempts['payload_path'] += 1
        raise RuntimeError('bulk archive payload read refused')
    check_time()
    return raw_read(root, relative, *args, **kwargs)


def guarded_blob(provider, oid, *args, **kwargs):
    if oid in forbidden_oids:
        attempts['payload_blob'] += 1
        raise RuntimeError('bulk archive blob read refused')
    check_time()
    return raw_blob(provider, oid, *args, **kwargs)


def local(relative):
    return archive._local(R, relative, archive.MAX_INDEX_BYTES)[0]


def value(relative):
    return archive._json(local(relative))


def logical(db):
    check_time()
    return {name: digest(db.execute('SELECT * FROM ' + name + ' ORDER BY ' + order).fetchall())
            for name, order in ORDER.items()}


def queue_copy():
    check_time()
    archive._path(R, 'state/queue.sqlite')
    live = sqlite3.connect((R / 'state/queue.sqlite').as_uri() + '?mode=ro', uri=True, timeout=1)
    memory = sqlite3.connect(':memory:')
    try:
        live.execute('PRAGMA query_only=ON')
        memory.execute('PRAGMA temp_store=MEMORY')
        require(memory.execute('PRAGMA temp_store').fetchone() == (2,), 'RAM temporary storage required')
        page_count = live.execute('PRAGMA page_count').fetchone()[0]
        page_size = live.execute('PRAGMA page_size').fetchone()[0]
        require(type(page_count) is int and type(page_size) is int and page_count > 0 and page_size > 0
                and page_count * page_size <= archive.MAX_INDEX_BYTES, 'RAM queue exceeds approved metadata bound')
        def progress(status, remaining, total):
            check_time()
            require(type(total) is int and 0 < total * page_size <= archive.MAX_INDEX_BYTES,
                    'RAM queue grew beyond approved metadata bound')
        live.backup(memory, pages=64, progress=progress, sleep=0.01)
        require(memory.execute('PRAGMA quick_check').fetchall() == [('ok',)], 'RAM queue quick_check failed')
        return memory
    except BaseException:
        memory.close()
        raise
    finally:
        live.close()


def counts(db, expected_seen):
    require(db.execute('SELECT COUNT(*),SUM(accepted),SUM(quarantined),SUM(status!="done") FROM jobs').fetchone()
            == (105, 3124, 0, 0), 'canonical count/status/quarantine mismatch')
    actual = {split: (groups, accepted, quarantined) for split, groups, accepted, quarantined in
              db.execute('SELECT split,COUNT(*),SUM(accepted),SUM(quarantined) FROM jobs GROUP BY split')}
    require(actual == SPLITS, 'canonical split counts changed')
    seen = db.execute('SELECT id,semantic,parent,template,split FROM seen ORDER BY id').fetchall()
    require(seen == sorted(expected_seen.values()), 'SQL/protected manifest identities differ')
    for column in ('semantic', 'parent', 'template'):
        require(db.execute('SELECT COUNT(*) FROM (SELECT ' + column +
                           ' FROM seen GROUP BY ' + column + ' HAVING COUNT(DISTINCT split)>1)').fetchone()[0]
                == 0, 'protected split overlap')
    require(db.execute('SELECT status,COUNT(*) FROM rendered_jobs GROUP BY status ORDER BY status').fetchall()
            == [('done', 42), ('pending', 87), ('superseded', 268)], 'render queue inventory changed')


def guard(pause, receipt, controls):
    check_time()

    require(local('state/PAUSED.json') == pause, 'owned PAUSED bytes changed')
    current = source_hashes()

            'current source69 changed')
    for relative, expected in controls.items():
        require(hashlib.sha256(local(relative)).hexdigest() == expected, 'frozen data/control bytes changed')


def phase(name, operation):
    before = time.monotonic()
    result = operation()
    check_time()
    phases[name] = round(time.monotonic() - before, 3)
    return result
