"""Bounded resumable publication of derived views; canonical gold stays immutable."""
import json
import time
from pathlib import Path

from ..decision_labels.rendering import (EXPORT_VERSION, RENDERING_POLICY_SET_VERSION,
                                         SUPPORTED_RENDERING_VERSIONS,
                                         VIEWS, export_rendered_dataset, _storage_guard, _archive_call)
from .common import canonical, digest, file_hash, immutable_bytes, source_hashes
from . import remote_archive


def settings(config):
    views = config.get('rendered_views', [])
    variants = config.get('rendered_variant_budget', 3)
    groups = config.get('rendered_groups_per_cycle', 12)
    if (not isinstance(views, list) or len(set(views)) != len(views)
            or any(view not in VIEWS for view in views)):
        raise ValueError('invalid rendered view configuration')
    if type(variants) is not int or not 1 <= variants <= 3:
        raise ValueError('invalid rendered variant budget')
    if type(groups) is not int or not 1 <= groups <= 16:
        raise ValueError('invalid bounded rendered group budget')
    return {'views': views, 'variant_budget': variants, 'groups_per_cycle': groups}


def ensure_queue(db):
    db.execute('CREATE TABLE IF NOT EXISTS rendered_jobs ('
               'source_manifest TEXT, source_sha256 TEXT, spec_sha256 TEXT, '
               'seed INTEGER, split TEXT, status TEXT, receipt TEXT, '
               'PRIMARY KEY(source_manifest,source_sha256,spec_sha256))')
    db.commit()


def _path(root, relative, prefix):
    root = Path(root).resolve()
    path = Path(relative)
    if path.is_absolute() or '..' in path.parts or path.parts[:len(prefix)] != tuple(prefix):
        raise ValueError('unmanaged rendered inventory path')
    full = root / path
    if full.is_symlink() or full.resolve() != full:
        raise ValueError('rendered inventory path must not traverse symlinks')
    return full


def _batch(root, relative):
    root = Path(root).resolve()
    path = _path(root, relative, ('manifests',))
    if path.parent != root / 'manifests':
        raise ValueError('canonical batch receipt must be directly under manifests')
    batch = json.loads(_archive_call('read_text',root,relative))
    split, seed = batch['split'], batch['seed']
    if split not in ('train', 'dev', 'ood'):
        raise ValueError('sealed or unknown batch cannot enter rendered queue')
    if path.name != f'batch-{split}-{seed}.json':
        raise ValueError('canonical batch receipt filename mismatch')
    shard = _path(root, batch['shard'], ('data', 'accepted'))
    if shard.parent != root / 'data/accepted' or shard.name != f'{split}-{seed}.jsonl':
        raise ValueError('canonical batch shard path mismatch')
    info=_archive_call('payload_info',root,batch['shard'])
    if (info['sha256'] != batch['sha256'] or info['bytes'] != batch['bytes']
            or batch.get('audit', {}).get('verdict') not in ('PASS', 'PARTIAL_QUARANTINE')
            or (batch.get('audit', {}).get('verdict') == 'PARTIAL_QUARANTINE' and batch.get('quarantined', 0) <= 0)
            or type(batch['accepted']) is not int or not 1 <= batch['accepted'] <= 12000):
        raise ValueError('canonical batch commitment mismatch')
    return batch


def enqueue(db, root, config):
    """Recover jobs from committed nonsealed batch receipts, without sealed gold."""
    root = Path(root).resolve()
    ensure_queue(db)
    cfg = settings(config)
    if not cfg['views']:
        return 0
    spec = {'export_version': EXPORT_VERSION, 'rendering_version': RENDERING_POLICY_SET_VERSION,
            'supported_rendering_versions':list(SUPPORTED_RENDERING_VERSIONS),
            'views': cfg['views'], 'variant_budget': cfg['variant_budget'],
            'code_sha256': digest(source_hashes())}
    spec_hash = digest(spec)
    for relative in _archive_call('iter_paths',root,'manifests/batch-*.json'):
        path = _path(root, relative, ('manifests',))
        # Split identification is by receipt data, not reading evaluation rows.
        metadata = json.loads(_archive_call('read_text',root,relative))
        if metadata.get('split') == 'sealed' or metadata.get('accepted') == 0:
            continue
        batch = _batch(root, relative)
        db.execute('INSERT OR IGNORE INTO rendered_jobs VALUES (?,?,?,?,?,?,?)',
                   (relative, _archive_call('payload_info',root,relative)['sha256'], spec_hash, batch['seed'], batch['split'], 'pending', None))
    # A reviewed executable upgrade replaces pending derived work with the new
    # specification; old completed receipts/bytes remain authoritative.
    db.execute('UPDATE rendered_jobs SET status="superseded" WHERE status="pending" AND spec_sha256!=?',(spec_hash,))
    db.commit()
    return db.execute('SELECT COUNT(*) FROM rendered_jobs WHERE status="pending"').fetchone()[0]


def validate_receipt(root, relative, *, use_controls=True):
    """Authenticate only committed exports; unfinished directories are excluded."""
    root = Path(root).resolve()
    path = _path(root, relative, ('manifests',))
    if path.parent != root / 'manifests' or not path.name.startswith('render-batch-'):
        raise ValueError('unmanaged rendered batch receipt')
    receipt = json.loads(_archive_call('read_text',root,relative))
    batch = _batch(root, receipt['source_manifest'])
    if (_archive_call('payload_info',root,receipt['source_manifest'])['sha256'] != receipt['source_sha256']
            or receipt['source_shard_sha256'] != batch['sha256']
            or receipt['split'] != batch['split'] or receipt['seed'] != batch['seed']
            or digest(receipt['specification']) != receipt['spec_sha256']
            or digest(receipt['code_hashes']) != receipt['specification']['code_sha256']):
        raise ValueError('rendered receipt canonical source/specification mismatch')
    expected_views = receipt['specification']['views']
    if [item['view'] for item in receipt['datasets']] != expected_views:
        raise ValueError('rendered receipt view inventory mismatch')
    # Unqualified/new exports retain the full raw path. Controls never supply
    # deletion custody and builder derivation explicitly bypasses this branch.
    projected = None
    if use_controls:
        from .render_controls import try_projection
        projected = try_projection(root, relative, receipt, batch)
    counts = {}
    seen = set()
    for position, item in enumerate(receipt['datasets']):
        manifest_path = _path(root, item['manifest_path'], ('data', 'rendered'))
        if _archive_call('payload_info',root,item['manifest_path'])['sha256'] != item['manifest_sha256']:
            raise ValueError('rendered dataset manifest digest mismatch')
        manifest = (projected[position] if projected is not None else
                    json.loads(_archive_call('read_text',root,item['manifest_path'])))
        if (manifest['manifest_path'] != item['manifest_path']
                or manifest['dataset_id'] in seen or manifest['dataset_id'] != item['dataset_id']
                or manifest['view'] != item['view'] or manifest['split'] != receipt['split']
                or manifest['source_records_examined'] != batch['accepted']
                or manifest['eligible_source_records'] != item['eligible_source_records']
                or manifest['code_hashes'] != receipt['code_hashes']
                or manifest['variant_budget'] != receipt['specification']['variant_budget']
                or manifest['export_version'] != receipt['specification']['export_version']
                or manifest['training_launched'] is not False
                or len(manifest['sources']) != 1):
            raise ValueError('rendered dataset identity or provenance mismatch')
        if 'supported_rendering_versions' in receipt['specification']:
            versions=manifest.get('rendering_versions',[])
            if (not versions or versions!=sorted(set(versions))
                    or any(v not in receipt['specification']['supported_rendering_versions'] for v in versions)
                    or manifest.get('rendering_policy_set_version')!=receipt['specification']['rendering_version']
                    or manifest['rendering_version']!=(versions[0] if len(versions)==1 else 'mixed')):
                raise ValueError('rendered policy version inventory mismatch')
        elif manifest['rendering_version']!=receipt['specification']['rendering_version']:
            raise ValueError('legacy rendered policy version mismatch')
        source = manifest['sources'][0]
        if (source['path'] != batch['shard'] or source['sha256'] != batch['sha256']
                or source['source_batch_manifest'] != receipt['source_manifest']
                or source['source_batch_manifest_sha256'] != receipt['source_sha256']
                or source['source_code_hashes'] != batch['code_hashes']
                or source['protected_index'] != batch['protected_index']):
            raise ValueError('rendered dataset source commitment mismatch')
        seen.add(manifest['dataset_id'])
        row_count = 0
        for shard in manifest['shards']:
            shard_path = _path(root, shard['path'], ('data', 'rendered'))
            info=_archive_call('payload_info',root,shard['path'])
            if (shard_path.parent != manifest_path.parent
                    or info['sha256'] != shard['sha256']
                    or info['bytes'] != shard['bytes']
                    or shard['bytes'] >= 8 * 1024 ** 2):
                raise ValueError('rendered shard digest/size/path mismatch')
            row_count += shard['rows']
        if row_count != manifest['rendered_rows'] or row_count != item['rendered_rows']:
            raise ValueError('rendered row count commitment mismatch')
        counts[item['view']] = row_count
    return receipt, counts


def recover(db, root):
    """A completed immutable receipt repairs a database interrupted at commit."""
    root = Path(root).resolve()
    ensure_queue(db)
    done = [relative for (relative,) in db.execute('SELECT receipt FROM rendered_jobs WHERE status="done"').fetchall()]
    discovered = _archive_call('iter_paths',root,'manifests/render-batch-*.json')
    for relative in done:
        if relative not in discovered:
            validate_receipt(root, relative)
    for relative in discovered:
        try:
            receipt, _ = validate_receipt(root, relative)
        except BaseException:
            # Setup committed prior work. A previously completed receipt must
            # fail without retaining any of this recovery's authenticated prefix.
            if relative in done:
                db.rollback()
            raise
        db.execute('INSERT OR REPLACE INTO rendered_jobs VALUES (?,?,?,?,?,?,?)',
                   (receipt['source_manifest'], receipt['source_sha256'], receipt['spec_sha256'],
                    receipt['seed'], receipt['split'], 'done', relative))
    db.commit()


def drain(db, root, config, deadline):
    """Finish at most the configured group budget and checkpoint each group."""
    root = Path(root).resolve()
    cfg = settings(config)
    if not cfg['views']:
        return []
    code = source_hashes()
    spec = {'export_version': EXPORT_VERSION, 'rendering_version': RENDERING_POLICY_SET_VERSION,
            'supported_rendering_versions':list(SUPPORTED_RENDERING_VERSIONS),
            'views': cfg['views'], 'variant_budget': cfg['variant_budget'],
            'code_sha256': digest(code)}
    spec_hash = digest(spec)
    jobs = db.execute('SELECT source_manifest,source_sha256,seed,split FROM rendered_jobs '
                      'WHERE status!="done" AND spec_sha256=? ORDER BY seed DESC LIMIT ?',
                      (spec_hash, cfg['groups_per_cycle'])).fetchall()
    completed = []
    for relative, source_hash, seed, split in jobs:
        if time.monotonic() >= deadline:
            break
        batch = _batch(root, relative)
        if _archive_call('payload_info',root,relative)['sha256'] != source_hash:
            raise ValueError('queued canonical source receipt changed')
        datasets = []
        for view in cfg['views']:
            if time.monotonic() >= deadline:
                return completed  # Partial exports stay pending and replay deterministically.
            try:
                manifest = export_rendered_dataset(root, split, view, batch['accepted'],
                                                   cfg['variant_budget'], source_paths=[root / batch['shard']],
                                                   deadline=deadline)
            except remote_archive.ReadDeadlineExceeded:
                return completed  # Only finished immutable receipts advance jobs.
            datasets.append({'view': view, 'dataset_id': manifest['dataset_id'],
                             'manifest_path': manifest['manifest_path'],
                             'manifest_sha256': file_hash(root / manifest['manifest_path']),
                             'rendered_rows': manifest['rendered_rows'],
                             'eligible_source_records': manifest['eligible_source_records']})
        if source_hashes() != code:
            raise RuntimeError('reviewed source changed during rendered batch publication')
        receipt = {'version': 'gcdl-rendered-batch-1', 'seed': seed, 'split': split,
                   'source_manifest': relative, 'source_sha256': source_hash,
                   'source_shard_sha256': batch['sha256'], 'specification': spec,
                   'spec_sha256': spec_hash, 'code_hashes': code, 'datasets': datasets,
                   'training_launched': False}
        receipt_path = root / 'manifests' / f'render-batch-{split}-{seed}-{spec_hash[:16]}.json'
        _path(root, str(receipt_path.relative_to(root)), ('manifests',))
        if receipt_path.with_name(receipt_path.name + '.tmp').is_symlink():
            raise ValueError('rendered receipt temporary path cannot be a symlink')
        raw_receipt = (canonical(receipt) + '\n').encode()
        if time.monotonic()>=deadline:return completed
        _storage_guard(root, 0 if receipt_path.exists() else len(raw_receipt))
        immutable_bytes(receipt_path, raw_receipt)
        receipt_relative = str(receipt_path.relative_to(root))
        validate_receipt(root, receipt_relative)
        db.execute('UPDATE rendered_jobs SET status="done",receipt=? WHERE '
                   'source_manifest=? AND source_sha256=? AND spec_sha256=?',
                   (receipt_relative, relative, source_hash, spec_hash))
        db.commit()
        completed.append(receipt_relative)
    return completed


def inventory(root):
    """Counts are rendered rows and source records, never additional worlds."""
    root = Path(root).resolve()
    counts, eligible = {}, {}
    groups = 0
    specifications = set()
    for relative in _archive_call('iter_paths',root,'manifests/render-batch-*.json'):
        receipt, rows = validate_receipt(root, relative)
        groups += 1
        specifications.add(receipt['spec_sha256'])
        for item in receipt['datasets']:
            key = receipt['split'] + '/' + item['view']
            counts[key] = counts.get(key, 0) + rows[item['view']]
            eligible[key] = eligible.get(key, 0) + item['eligible_source_records']
    return {'completed_source_group_exports': groups, 'specifications': sorted(specifications),
            'rendered_rows_by_split_view': counts, 'eligible_source_records_by_split_view': eligible,
            'count_interpretation': 'derived surface rows; multiple views/variants repeat canonical source worlds',
            'training_launched': False}
