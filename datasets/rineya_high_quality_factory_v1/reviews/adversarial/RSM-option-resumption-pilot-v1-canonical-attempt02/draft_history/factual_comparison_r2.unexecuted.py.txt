def canon(v):return json.dumps(v,sort_keys=True,separators=(',',':'),ensure_ascii=False,allow_nan=False)
def semantic_hash(v):return hashlib.sha256(canon(v).encode()).hexdigest()
def bind(p):
    p=Path(p);raw=p.read_bytes()
    return {'path':str(p.relative_to(ROOT)) if p.is_relative_to(ROOT) else str(p),'size_bytes':len(raw),'sha256':hashlib.sha256(raw).hexdigest()}
def record_hash(row):return semantic_hash({k:v for k,v in row.items() if k!='record_hash'})
def write(name,value,jsonl=False):
    p=OUT/name
    with p.open('x') as f:
        if jsonl:
            for r in value:f.write(canon(r)+'\n')
        else:json.dump(value,f,indent=2,ensure_ascii=False,allow_nan=False);f.write('\n')
    if p.stat().st_size>25000000:raise ValueError('artifact exceeds cap')
    return bind(p)
def assert_true(v,msg):

def safe_file(raw):

    p=PurePosixPath(raw)
    assert_true(not p.is_absolute() and p.as_posix()==raw and all(x not in {'.','..'} for x in p.parts),'safe relative path')

    assert_true(actual.is_relative_to(ROOT) and actual.is_file(),'existing bound file')
    return actual
def validate_schema(value,schema):
    """Interpret the actual public canonical schema keywords, no private gate.
    Strict integers excludebool, consistent with task and public metadata contract.
    """
    types={'object':dict,'array':list,'string':str,'integer':int,'boolean':bool,'null':type(None)}
    if 'type' in schema:assert_true(type(value) is types[schema['type']],'schema type')
    if 'enum' in schema:assert_true(any(type(value) is type(v) and value==v for v in schema['enum']),'schema enum')
    if 'minimum' in schema:assert_true(value>=schema['minimum'],'schema minimum')
    if 'maximum' in schema:assert_true(value<=schema['maximum'],'schema maximum')
    if 'pattern' in schema:assert_true(re.search(schema['pattern'],value) is not None,'schema pattern')
    if type(value) is dict:
        assert_true(set(schema.get('required',[]))<=set(value),'schema required')
        if schema.get('additionalProperties') is False:assert_true(set(value)<=set(schema.get('properties',{})),'schema extra keys')
        for k,sub in schema.get('properties',{}).items():
            if k in value:validate_schema(value[k],sub)
    if type(value) is list and 'items' in schema:
        for v in value:validate_schema(v,schema['items'])
    return True
def path_value(value,path):
    assert_true(type(path) is list,'locator list')
    current=value
    for key in path:
        if type(current) is dict:assert_true(type(key) is str and key in current,'literal object locator');current=current[key]
        elif type(current) is list:assert_true(type(key) is int and 0<=key<len(current),'typed list index');current=current[key]
        else:raise ValueError('locator traverses scalar')
    return current
def set_leaf(value,path,new):
    assert_true(path,'nonempty leaf path');parent=path_value(value,path[:-1]);parent[path[-1]]=new

def lift_legacy(task):
    """Lift only supplied target-free legacy primitives/main into common grammar."""
    assert_true(task['kind']=='skills','legacy kind')
    root='__own_legacy_main__'
    while root in task['procedures']:root+='x'
    def body(old):
        out=[]
        for ins in old:
            op=ins['op']
            if op in {'add','scale'}:out.append({'op':op,'register':ins['register'],'value':{'const':ins['value']}})
            elif op=='swap':out.append({'op':op})
            elif op=='call':out.append({'op':op,'option':ins['name'],'args':{}})
            elif op=='if':
                t=ins['test'];out.append({'op':op,'test':{'all':[{'left':{'reg':t['register']},'cmp':t['cmp'],'right':{'const':t['value']}}]},'then':body(ins['then']),'else':body(ins['else'])})
            else:raise ValueError('unsupported public legacy instruction')
        return out
    def definition(instructions):return {'params':[],'initiation':{'all':[]},'success':{'all':[]},'body':body(instructions)}
    options={name:definition(instructions) for name,instructions in task['procedures'].items()};options[root]=definition(task['main'])

def anchored_closure(root,out,args,phase):
    """External caller anchors → fullcurrent/rawclosure → actual runtime files.
    File-backed sources and existing caches must both be frozen, even where
    CPython reports a frozen origin. Namespace directories are only descriptors.
    Built-in/frozen code is limited by the bound resolvedPython binary, not a
    claim that a source/cache path was executed as Python text.
    """
    def rawhash(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
    if rawhash(out/'RUN_CONFIG.json')!=args.expected_config_sha256:raise ValueError('external config SHA mismatch')
    if rawhash(out/'SOURCE_FREEZE.json')!=args.expected_source_freeze_sha256:raise ValueError('external sourcefreeze SHA mismatch')
    if rawhash(args.grant)!=args.expected_grant_sha256:raise ValueError('external grant SHA mismatch')
    grant=json.loads(Path(args.grant).read_text())
    if grant.get('authorized_by')!='/root' or grant.get('phase')!=phase:raise ValueError('explicit stage execution grant')
    if grant.get('config_sha256')!=args.expected_config_sha256 or grant.get('source_freeze_sha256')!=args.expected_source_freeze_sha256:raise ValueError('grant anchors')
    freeze=json.loads((out/'SOURCE_FREEZE.json').read_text())
    actual_python=str(Path(sys.executable).resolve())
    if actual_python!=freeze['resolved_python']:raise ValueError('resolvedPython path mismatch')
    bound={};pairs=[]
    for row in freeze['operative_bindings']:
        p=Path(row['path']);p=p if p.is_absolute() else root/p
        p=p.resolve();archive=root/row['archive'];current=p.read_bytes();stored=archive.read_bytes()
        if len(current)!=row['size_bytes'] or len(stored)!=row['size_bytes'] or current!=stored:raise ValueError('current/raw exactpair '+row['path'])
        if hashlib.sha256(current).hexdigest()!=row['sha256'] or hashlib.sha256(stored).hexdigest()!=row['sha256']:raise ValueError('current/raw SHA '+row['path'])
        bound[str(p)]={'sha256':row['sha256'],'size_bytes':row['size_bytes']}
        pairs.append({'path':row['path'],'sha256':row['sha256'],'size_bytes':row['size_bytes'],'archive':row['archive']})
    if actual_python not in bound:raise ValueError('resolvedPython bytes unbound')
    modules=[]
    for name,module in sorted(sys.modules.items()):
        spec=getattr(module,'__spec__',None);origin=getattr(spec,'origin',None)
        concrete=[]
        source=getattr(module,'__file__',None);cache=getattr(module,'__cached__',None)
        for kind,value in [('source_or_extension',source),('cache',cache)]:
            if not value:continue
            path=Path(value).resolve()
            if not path.is_file():
                if kind=='source_or_extension':raise ValueError('loaded file missing '+name)
                continue
            key=str(path)
            if key not in bound:raise ValueError('new/unbound concrete runtime '+name+' '+kind+' '+key)
            raw=path.read_bytes();actual={'sha256':hashlib.sha256(raw).hexdigest(),'size_bytes':len(raw)}
            if actual!=bound[key]:raise ValueError('loaded runtime SHA/size mismatch '+name+' '+kind)
            concrete.append({'kind':kind,'path':key,**actual})
        namespace=list(getattr(module,'__path__',[])) if origin is None and hasattr(module,'__path__') else []
        classification='namespace' if namespace else 'builtin' if origin=='built-in' else 'frozen' if origin=='frozen' else 'file-backed' if concrete else 'no-concrete-file'
        modules.append({'module':name,'classification':classification,'origin':origin,'concrete_bindings':concrete,
                        'namespace_paths':namespace,'boundary':'directory locations only; contents not certified' if namespace else 'boundPython handles builtin/frozen origins; concrete fields independently bound'})
    return {'external_config_sha256':args.expected_config_sha256,'external_source_freeze_sha256':args.expected_source_freeze_sha256,
            'external_grant_sha256':args.expected_grant_sha256,'resolved_python':actual_python,'all_exact_raw_pairs_verified':pairs,
            'loaded_runtime_modules':modules,'unbound_concrete_file_modules':[]}
